# ---- batchtime.cpp ----
# Built from the repository root with:
#   c++ -std=c++17 -O3 -DNDEBUG -DAZ_USE_BLAS -DACCELERATE_NEW_LAPACK -Iinclude \
#       batchtime.cpp build-blas/libazgo.a -framework Accelerate -o batchtime
# and run as ./batchtime from the repository root (it reads configs/5x5.cfg and configs/9x9.cfg).
// One-off: evals/s of the preset networks at batch 1 vs batch 8, one thread.
#include <chrono>
#include <cstdio>
#include <memory>
#include <vector>
#include "azgo/backend.hpp"
#include "azgo/config.hpp"
#include "azgo/network.hpp"
#include "azgo/rng.hpp"
using Clock = std::chrono::steady_clock;
static double measure(const azgo::NetConfig& nc, int batch, double seconds) {
    auto backend = azgo::makeBackend("blas", 1);
    const azgo::Network net(nc, 1);
    azgo::Workspace ws(nc, batch, *backend);
    azgo::Rng rng(7);
    std::vector<float> input(static_cast<size_t>(batch) * nc.planes() * nc.area());
    for (float& x : input) x = static_cast<float>(rng.below(2));
    std::vector<float> logits(static_cast<size_t>(batch) * nc.actions()), values(batch);
    net.infer(ws, input.data(), batch, logits.data(), values.data());
    long long calls = 0; auto start = Clock::now(); double el = 0;
    do { net.infer(ws, input.data(), batch, logits.data(), values.data()); ++calls;
         el = std::chrono::duration<double>(Clock::now() - start).count(); } while (el < seconds);
    return double(calls) * batch / el;
}
int main(int argc, char** argv) {
    for (const char* p : {"configs/5x5.cfg", "configs/9x9.cfg"}) {
        const azgo::NetConfig nc = azgo::loadConfigFile(p).netConfig();
        for (int rep = 0; rep < 3; ++rep) {
            double b1 = measure(nc, 1, 3.0), b8 = measure(nc, 8, 3.0);
            std::printf("%dx%d  %d x %d  blas 1 thread  batch1 %8.0f evals/s  batch8 %8.0f evals/s  ratio %.2f\n",
                        nc.boardSize, nc.boardSize, nc.blocks, nc.filters, b1, b8, b8 / b1);
            std::fflush(stdout);
        }
    }
}

# Scratch program: one call of each conv gemm shape unsplit on one thread,
# then cut by rows (bands of H rows, remainder merged) or by columns (blocks
# of W columns) on the pool; im2col, col2im and batch norm without and with
# the pool.
#   6x96 B=256 9x9, 18 threads (ms)
#     fwd  unsplit 1T    3.45
#     dW   unsplit 1T    2.74
#     dX   unsplit 1T    4.06
#     rows H=16  fwd    4.72  dW    3.75  dX    1.41
#     rows H=32  fwd    3.83  dW    3.19  dX    1.57
#     rows H=48  fwd    3.66  dW    2.98  dX    1.52
#     cols W=256  fwd    1.41  dX    1.52
#     cols W=512  fwd    1.48  dX    1.59
#     cols W=1024 fwd    1.59  dX    1.55
#     cols W=2048 fwd    1.93  dX    1.84
#     im2col 1T    4.76   pooled    0.41
#     col2im 1T    2.53   pooled    0.36
#     BN fwd 1T    2.43   pooled    0.22
#     BN bwd 1T    1.30   pooled    0.16
# Row bands re-read the whole column matrix per band, and on Accelerate the
# forward and dW bands were slower than one call; the first version, with
# 16-row bands for every conv gemm, gave 6x64 83.9 ms and 6x96 141.7 ms on
# 18 threads. The final code cuts the forward and dX gemms into 512-column
# blocks and runs dW as one call beside them (the naive backend cuts dW
# into 16-row bands after packing once).
#
# ---- Determinism ----
# Accelerate, column blocks and row bands against the unsplit call: a sweep
# of ~11,000 conv shapes (boards 5-19, 8-256 filters, batches 1-256) found
# every column block (128-1024 columns, >= 8 rows) and row band (8-32 rows)
# bitwise equal; single-row bands and column splits of 1-2-row products and
# of some small-batch dW gemms were not, and are not used. Old (main) against
# new train binary, byte-identical checkpoints, state.bin, results.csv (and
# elo.csv, references.csv, log.csv without the timing columns): 5x5 preset
# 3 generations (40 games, 50 steps) naive and BLAS; 9x9 at batch 256, 2
# generations, BLAS 6x64 and 6x96 and naive 6x64, each with train_threads 0
# and 1; the 19x19 preset's 10x128 network, 1 generation (4 games, 3 steps);
# and the full 5x5 preset with BLAS for 30 generations (checkpoints 1-30
# identical; an earlier version of the branch, with 16-row bands).
